@echo off
title Questmates installer
powershell -NoProfile -ExecutionPolicy Bypass -Command "$f='%~f0'; iex ((Get-Content -LiteralPath $f -Raw) -split '#PS_START#')[2]"
echo.
pause
exit /b
#PS_START#
$ErrorActionPreference = 'Stop'
Write-Host ''
Write-Host '  Questmates installer (WoW Forever)' -ForegroundColor Yellow
Write-Host ''

function Test-WowRoot($p) { $p -and (Test-Path -LiteralPath (Join-Path $p '_forever_')) }

$cands = New-Object System.Collections.Generic.List[string]
foreach ($key in 'HKLM:\SOFTWARE\WOW6432Node\Blizzard Entertainment\World of Warcraft', 'HKLM:\SOFTWARE\Blizzard Entertainment\World of Warcraft') {
  try { $ip = (Get-ItemProperty -LiteralPath $key -ErrorAction Stop).InstallPath; if ($ip) { $cands.Add((Split-Path ($ip.TrimEnd('\')) -Parent)); $cands.Add($ip.TrimEnd('\')) } } catch {}
}
foreach ($u in 'HKLM:\SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall\*', 'HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall\*') {
  try { Get-ItemProperty $u -ErrorAction SilentlyContinue | Where-Object { $_.DisplayName -like 'World of Warcraft*' -and $_.InstallLocation } | ForEach-Object { $cands.Add($_.InstallLocation.TrimEnd('\')) } } catch {}
}
foreach ($d in (Get-PSDrive -PSProvider FileSystem | ForEach-Object { $_.Root })) {
  foreach ($sub in 'Program Files (x86)\World of Warcraft', 'Program Files\World of Warcraft', 'World of Warcraft', 'Games\World of Warcraft', 'Battle.net\World of Warcraft', 'Blizzard\World of Warcraft') { $cands.Add((Join-Path $d $sub)) }
}
if ($env:QM_TEST_ROOT) { $cands.Clear(); $cands.Add($env:QM_TEST_ROOT) }

$root = $null
foreach ($c in $cands) { if (Test-WowRoot $c) { $root = $c; break } }

if (-not $root) {
  Write-Host '  Could not find WoW Forever automatically - please pick your "World of Warcraft" folder.'
  Add-Type -AssemblyName System.Windows.Forms
  $dlg = New-Object System.Windows.Forms.FolderBrowserDialog
  $dlg.Description = 'Pick your "World of Warcraft" folder (the one that contains _forever_)'
  if ($dlg.ShowDialog() -eq 'OK') {
    $p = $dlg.SelectedPath
    if (Test-WowRoot $p) { $root = $p }
    elseif ((Split-Path $p -Leaf) -eq '_forever_') { $root = Split-Path $p -Parent }
  }
}
if (-not $root) { Write-Host '  No WoW Forever folder found - nothing was installed.' -ForegroundColor Red; return }

$addons = Join-Path $root '_forever_\Interface\AddOns'
$target = Join-Path $addons 'Questmates'
$text = Get-Content -LiteralPath $f -Raw
if ((Test-Path $target) -and (Get-Item $target).LinkType) {
  Write-Host '  This PC has the Questmates developer copy (a linked folder) - leaving it alone.' -ForegroundColor Yellow
  return
}

# 1. the addon: newest release from GitHub, or the copy inside this file when offline
$source = 'the copy inside this installer'
try {
  New-Item -ItemType Directory -Force -Path $addons | Out-Null
  $tmp = Join-Path $env:TEMP ('Questmates_' + [guid]::NewGuid().ToString('N'))
  New-Item -ItemType Directory -Force $tmp | Out-Null
  $zip = Join-Path $tmp 'Questmates.zip'
  $got = $false
  if (-not $env:QM_OFFLINE) {
    try {
      [Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
      Invoke-WebRequest -Uri 'https://github.com/Bjornaman2/questmates/releases/latest/download/Questmates.zip' -OutFile $zip -UseBasicParsing -TimeoutSec 60
      $got = $true; $source = 'GitHub (newest version)'
    } catch { $got = $false }
  }
  if (-not $got) {
    $b64 = ($text -split ('#' + 'QMDATA#'))[1].Trim()
    [IO.File]::WriteAllBytes($zip, [Convert]::FromBase64String($b64))
  }
  Expand-Archive -LiteralPath $zip -DestinationPath (Join-Path $tmp 'x') -Force
  if (-not (Test-Path (Join-Path $tmp 'x\Questmates\Questmates.toc'))) { throw 'the download looks broken' }
  if (Test-Path $target) { Remove-Item -LiteralPath $target -Recurse -Force }
  Copy-Item (Join-Path $tmp 'x\Questmates') $target -Recurse -Force
  Remove-Item $tmp -Recurse -Force
} catch {
  Write-Host ('  Install failed: ' + $_.Exception.Message) -ForegroundColor Red
  Write-Host '  Try again: right-click this file and choose "Run as administrator".'
  return
}
$ver = (Select-String -Path (Join-Path $target 'Questmates.toc') -Pattern '^## Version:\s*(\S+)').Matches[0].Groups[1].Value
Write-Host ('  Installed Questmates ' + $ver + ' from ' + $source + ':') -ForegroundColor Green
Write-Host ('  ' + $target)
Write-Host ''

# 2. automatic updates: a small script that checks GitHub at sign-in and every 3 hours
$updated = $false
try {
  $home_ = Join-Path $env:LOCALAPPDATA 'Questmates'
  New-Item -ItemType Directory -Force $home_ | Out-Null
  $upd = ($text -split ('#' + 'UPD_START#'))[1]
  $upd = ($upd -split ('#' + 'UPD_END#'))[0].Trim()
  [IO.File]::WriteAllText((Join-Path $home_ 'update.ps1'), $upd, (New-Object Text.UTF8Encoding($true)))
  (@{ addons = $addons } | ConvertTo-Json) | Out-File -Encoding utf8 (Join-Path $home_ 'config.json')
  $vbs = 'Set sh = CreateObject("WScript.Shell")' + "`r`n" +
         'sh.Run "powershell.exe -NoProfile -ExecutionPolicy Bypass -WindowStyle Hidden -File """ & sh.ExpandEnvironmentStrings("%LOCALAPPDATA%") & "\Questmates\update.ps1""", 0, False'
  [IO.File]::WriteAllText((Join-Path $home_ 'run-hidden.vbs'), $vbs)
  if (-not $env:QM_NO_SCHEDULE) {
    $startup = Join-Path ([Environment]::GetFolderPath('Startup')) 'Questmates Updater.vbs'
    [IO.File]::WriteAllText($startup, $vbs)
    $tr = 'wscript.exe "' + (Join-Path $home_ 'run-hidden.vbs') + '"'
    schtasks.exe /Create /F /TN 'Questmates Updater' /SC HOURLY /MO 3 /TR $tr 2>$null | Out-Null
  }
  $updated = $true
} catch {
  Write-Host ('  Automatic updates could not be set up: ' + $_.Exception.Message) -ForegroundColor Yellow
}

if ($updated) {
  Write-Host '  Automatic updates are on: Questmates keeps itself up to date (checks at sign-in and every 3 hours).' -ForegroundColor Green
  Write-Host '  New versions load the next time you start WoW (or type /reload).'
}
Write-Host ''
if (Get-Process -Name 'Wow' -ErrorAction SilentlyContinue) { Write-Host '  WoW is running - restart the game (or /reload) to load Questmates.' -ForegroundColor Yellow }
else { Write-Host '  Start WoW Forever. Questmates is on (check AddOns at character select).' }
Write-Host '  In game: /qm opens the Party Quest Journal, /qm demo shows a preview.'
return
#PS_START#
